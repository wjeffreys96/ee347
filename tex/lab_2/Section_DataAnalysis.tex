\subsection{Methods of Calculation}
\begin{itemize}
	\item Z$_\|$ The resulting parallel impedance, if any, after adding the capacitive load
	      \begin{center}
		      \item Z$_{\|} =$ $\frac{
				      Z_{1}Z_{2}
			      }{
				      Z_1 + Z_2
			      }$
	      \end{center}

	\item P - The real power in watts
	      \begin{center}
		      P = $3VI^*$
	      \end{center}
\end{itemize}
