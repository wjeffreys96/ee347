\subsection{Open Circuit Test}
\begin{enumerate}
	\item Follow the lock-out/tag-out start-up procedure
	\item Energize the power supply
	\item Use the LVDAC to take measurements of $V_{OC}$, $I_{OC}$, $P_{OC}$, and $Q_{OC}$
	      \begin{itemize}
		      \item $V_{OC}$ - Measure the voltage across the secondary side terminals
		      \item $I_{OC}$ - Measure the current through the primary side terminals
		      \item $P_{OC}$ and $Q_{OC}$ - The LVDAC will calculate these from $V_{OC}$ and $I_{OC}$
	      \end{itemize}
	\item Calculate $R_{c}$ and $X_{m}$
	\item De-energize the power supply
	\item Follow the lock-out/tag-out shut-down procedure
\end{enumerate}
\subsection{Short Circuit Test}
\begin{enumerate}
	\item Follow the lock-out/tag-out start-up procedure
	\item Double-check to ensure that the 120 V variable source is initially set to 0 V (this is critical)
	\item Short the terminals on the secondary side
	\item Continuously monitor the primary- current
	\item Slowly adjust the power supply rheostat until primary current equals the rated current
	\item Double-check the current on the secondary, ensure it doesn't exceed 0.5 amps (rated)
	\item Use the LVDAC to take measurements of $V_{SC}$, $I_{SC}$, $P_{SC}$, and $Q_{SC}$
	      \begin{itemize}
		      \item $V_{SC}$ - Measure the voltage across the primary side terminals
		      \item $I_{SC}$ - Measure the current through the secondary side terminals
		      \item $P_{SC}$ and $Q_{SC}$ - The LVDAC will calculate these from $V_{SC}$ and $I_{SC}$
	      \end{itemize}
	\item Calculate $R_{eq}$ and $X_{eq}$
	\item De-energize the power supply
	\item Follow the lock-out/tag-out shut-down procedure
\end{enumerate}

\subsection{Calculations}
	\begin{center}
		\text{Open circuit test}
	\end{center}
	\begin{center}
		$I_m=\sqrt{I_{oc}^2-(\frac{P_{oc}}{V_{oc}})^2}=\sqrt{0.029^2-(\frac{2.15}{120.8})^2}=27.1mA$
	\end{center}
	\begin{center}
		$X_m=\frac{V_{oc}}{I_m}=\frac{120.8}{0.0271}=\boxed{5.28k\Omega}$
	\end{center}
	\begin{center}
		$R_c=\frac{V_{oc}^2}{P_{oc}}=\frac{120.8^2}{2.15}=\boxed{6.79k\Omega}$
	\end{center}
	\begin{center}
		\text{Short circuit test}
	\end{center}
	\begin{center}
		$I_{p,rated}=I_{sc}=\frac{P}{V_p}=\frac{60VA}{208V}=0.288A$
	\end{center}
	\begin{center}
		$R_{eq}=\frac{P_{sc}}{I_{sc}^2}=\frac{4.4}{0.5^2}=\boxed{17.6\Omega}$
	\end{center}
	\begin{center}
		$Z_{eq}=\frac{V_{sc}}{I_{sc}}=\frac{14.61V}{0.5A}=29.22\Omega$
	\end{center}
	\begin{center}
		$X_{eq}=\sqrt{Z_{eq}^2-R_{eq}^2}=\sqrt{29.22^2-17.6^2}=\boxed{23.3\Omega}$
	\end{center}
