\subsection{Calculations}
\begin{description}
	\item[$S_{3P}$] - Total 3 Phase Apparent Power Rating
	\begin{center}
		$S_{3p} = 3S_{\phi}$
	\end{center}
	\item[$V_{Bank}$] - Line Voltage of the Transformer Bank
	\begin{center}
	Wye/Delta: $V_{Bank} = V_{Rated}$
	\end{center}

	\item[$I_{Bank}$] - Line Current of the Transformer Bank
	\begin{center}
	Wye/Delta: $I_{Bank} = I_{Rated}$
	\end{center}

	\item[$V_{Phase}$] - Voltage in Each Phase of the Transformer Bank
	\begin{center}
	Wye: $V_{Phase} = \frac{V_{Bank}}{\sqrt{3}}$
	\end{center}
	\begin{center}
	Delta: $V_{Phase} = V_{Bank}$
	\end{center}

	\item[$I_{Phase}$] - Current in Each Phase of the Transformer Bank
	\begin{center}
	Wye: $I_{Phase} = I_{Bank}$
	\end{center}
	\begin{center}
	Delta: $I_{Phase} = \frac{I_{Bank}}{\sqrt{3}}$
	\end{center}
\end{description}
