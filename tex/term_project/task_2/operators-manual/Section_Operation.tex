\begin{enumerate}
	\item Plug in the power cord to an outlet providing 120\,V
	\item Remove the lock out locks on the lockout relay
	\item Press the red button at the top of the rack to turn power on to the circuit
	      \stepimg{Pictures/poweron.jpg}
	      \clearpage
	\item Verify no error lights are illuminated on the CLICK PLC and flip the switch on the CLICK PLC to the Run position. It should look like this:
	      \stepimg{Pictures/run_light.jpg}
	\item Observe the green light is illuminated
	      \stepimg{Pictures/green.jpg}
	      \clearpage
	\item Press the blue button
	      \stepimg{Pictures/press.jpg}
	\item Observe the red light is illuminated and the green light is no longer illuminated
	      \stepimg{Pictures/red.jpg}
	\item Turn off the power to the rack by turning the relay clockwise
	      \stepimg{Pictures/poweroff.jpg}
	\item Replace the locks on the lockout relay
	\item Unplug the rack
\end{enumerate}
