\newcommand{\stepimg}[1]{%
	\par\smallskip
	\begin{center}
		\includegraphics[width=0.4\textwidth,height=5cm,keepaspectratio]{#1}%
	\end{center}
	\par\smallskip
}

\begin{enumerate}
	\item Plug in the power cord to an outlet providing 120\,V
	\item Remove the lock out locks on the lockout relay
	\item Press the red button at the top of the rack to turn power on
	      to the circuit
	      \stepimg{Pictures/poweron.jpg}
	\item Verify no error lights are illuminated on the CLICK PLC and
	      flip the switch on the CLICK PLC to the Run position. It should
	      look like this:
	      \stepimg{Pictures/run_light.jpg}
	\item Observe the green lights associated with each switch gear is
	      illuminated indicating they are initially open. And the red
	      lights are dimmed.
	      \stepimg{Pictures/all_open.png}
	\item Press the blue button labeled "XFMR" to close the transformer
	      switch gear.
	      \stepimg{Pictures/xfmr_button.png}
	\item Watch the red light associated with the xfmr illuminate and
	      the green light dim.
	      \stepimg{Pictures/xfmr_closed.png}
	\item Press the blue button labeled "Bus" to close the bus switch gear.
	      \stepimg{Pictures/bus_button.png}
	\item Watch the red light associated with the bus illuminate and
	      the green light dim.
	      \stepimg{Pictures/bus_closed.png}
	\item Press the blue button labeled "F1" to close the F1 switch gear.
	      \stepimg{Pictures/f1_button.png}
	\item Watch the red light associated with the F1 illuminate and the
	      green light dim.
	      \stepimg{Pictures/f1_closed.png}
	\item Press the blue button labeled "F2" to close the F2 switch gear.
	      \stepimg{Pictures/f2_button.png}
	\item Watch the red light associated with the F2 illuminate and the
	      green light dim.
	      \stepimg{Pictures/all_closed.png}
	\item Turn on the VFD using the button on the front and turn the nob to change the speed of the motor.
	      \stepimg{Pictures/vfd.png}
	\item Turn off the power to the rack by turning the relay clockwise
	      \stepimg{Pictures/poweroff.jpg}
	\item Replace the locks on the lockout relay
	\item Unplug the rack
\end{enumerate}
